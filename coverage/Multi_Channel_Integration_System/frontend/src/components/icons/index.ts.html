  return defineComponent<IconProps>({
    name: 'Icon',
    props: {
      size: {
        type: [Number, String],
        default: 24
      },
      strokeWidth: {
        type: [Number, String],
        default: 2
      },
      class: {
        type: String,
        default: ''
      }
    },
    setup(props) {
      return () => h('svg', {
        width: String(props.size || 24),
        height: String(props.size || 24),
        viewBox,
        fill: 'none',
        stroke: 'currentColor',
        'stroke-width': String(props.strokeWidth || 2),
        'stroke-linecap': 'round',
        'stroke-linejoin': 'round',
        class: props.class || ''
      }, [
        h('path', { d: path })
      ])
    }
  })
}
